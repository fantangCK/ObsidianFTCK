```cpp title:隐式构造函数
#include <iostream>
#include <string>

class Entity
{
private;
	std::string m_Name;
	int m_Age;
public:
	Entity(const std::string& name)
		: m_Name(name), m_Age(-1) {}
		
	Entity(int age)
		: m_Name("Unknown"), m_Age(age) {}
};

void PrintEntity(const Entity& entity)
{
	//Do sth.
}

int main()
{
	PrintEntity(22); //22可以转换为Entity 然后调用可以接受int类型的构造函数
	PrintEntity(std::string("Cherno"));
						//此处直接用"Cherno"是不合法的
						//因为涉及了两次隐式转换 Entity和string
						//直接 传入Entity("Cherno")也行 然后在类中做隐式转换
	
	Entity a("Cherno");
	Entity b(22); //更加
	//等价 隐式构造函数
	Entity a = "Cherno";
	Entity b = 22; 
	
	std::cin.get();
}
```
