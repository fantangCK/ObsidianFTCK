```ad-note
title: enum-枚举
**enumeration** 数值集合
- 集合里面的东西只能是整型或者字符型
- 默认从0开始枚举

```

```cpp title:枚举
#include <iostream>

enum Example : char
{
	A = 5,B,C
	//如果没有指定值后面的元素依次递增5 6 7
	//可以强制指定A=5 B=2 C=7
	//默认从0开始枚举
};

int a = 0; b = 1; c = 2;

int main()
{
	Example value = B; //value取值只能是A B C对应的数字
	if (value == 1)
	{
		//Do sth.
	}
	std::cin.get();
}

```

```cpp title:案例
class Logy
{
public: 
	enum Level //Level本身不是一个命名空间 是枚举类
	{
		LevelError = 0, LevelWarning, LevelInfo
	}//这三者只存在于Log这个类中
private:
	Level m_LogLevel = LevelInfo;
public:
	void SetLevel(int level)
	 {
	 	m_LogLevel = level;
	 }
	 
	void Error(const char* message)
	{
		if(m_LogLevel >= LevelError)
			std::cout << "[ERROR]: " << message << std::endl;
	}	 
	
	void Info(const char* message)
	{
		if(m_LogLevel >= LevelInfo)
			std::cout << "[INFO]: " << message << std::endl;
	}	 
	 
	 void Warn(const char* message)
	 {
		 if(m_LogLevel >= LevelWarning)
		 	std::cout << "[WARNING]: " << message << std::endl;
	 }
};

int  main()
{
	Log log; //初始化
	log.SetLevel(Log::LevelError); //此处如果直接改为Error其实重名了 改为LevelError
	log.Warn("hello!");
	log.Error("hello!");
	log.Info("hello!");
	std::cin.get();
}
```
