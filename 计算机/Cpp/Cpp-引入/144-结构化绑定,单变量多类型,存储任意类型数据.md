```ad-note
title: 结构化绑定

结构化绑定仅限`c++17`之后的版本
更好的处理多返回值

```

```cpp title:结构化绑定
#include <iostream>
#include <string>
#include <tuple>

std:tuple<std:string, int>CreatePerson()
{	//这里也可以用pair键值对 这里就俩数据 而tuple允许更多类型和数量
	return { "Cherno", 24 }
}
int main()
{
	auto person CreatePerson(); //这里person只是一个容器
	std::string& name = std:get<0>(person);
	int age = std:get<1>(person);
	//以前的做法 只能用标号返回
	
	std::string name;
	int age;
	std::tie(name, age) = CreatePerson();
	//稍微好点的方法 但是也要先创建类型变量(不如结构体)
	
	auto[name, age] = CreatePerson(); //结构化绑定
	std::cout << name;
	//c++17带来的方法
}
```

```cpp title:旧处理方法
#include <iostream>
#include <fstream> //读写文件的库

std::string ReadFileAsString(const std::string& filepath, bool& outSuccess)
{										//或者传个参表示是否读取成功
	std::ifstream stream(filepath);
	if(stream)
	{
		std::string result;
		//read file
		stream.clsoe();
		outSuccess = true;
		return result
	}	//读取成功
	
	outSuccess = false;
	return std::string(); //读取不成功的一种处理方法
}

int main()
{
	bool fileOpenedSuccessfully;
	std::string data = ReadFileAsString("data.txt", fileOpenedSuccessfully);
	if (data != "")  {}
	if (fileOpenedSuccessfully)  {}
}

```
