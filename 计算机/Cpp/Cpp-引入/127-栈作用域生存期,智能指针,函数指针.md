```cpp title:作用域
	//int array[50]; //这样创建会因为作用域而被清理
	int array = new int[50]; //实际上在局部函数里面创建数组是很奇怪的
	return array;

{
	//Do sth.
}

class ScopedPtr //作用域指针类
{
private:
	Entity* m_Ptr;

		: m_Ptr(ptr)
	{
	}
	
	~ScopedPtr(Entity* ptr)
	{
		delete m_Ptr;
	}
};

}

{
	int* a = CreateArray();
```

```ad-note
title: 智能指针

不用释放堆内存

智能指针有几个类型
- unique_ptr 指针无法复制和传递 
- shared_ptr 有一个引用计数 当引用计数为0时全部释放
- weak_ptr 被赋值等等操作 不会增加引用计数
```

```cpp title:智能指针
#include <iostream>
#include <string>

#include <memory>
class Entity
{
public:
	
	Entity()
	{
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity()
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}

};

int main()
{
	{
		//std::unique_ptr<Entity> entity(new Entity())
		//更好的做法
		std::unique_ptr<Entity> entity = std::make_unique<Entity>();
		
		//std::unique_ptr<Entity> e0 = entity; //不能这么做 因为一旦复制 等到作用域结束后 这个内存也会被强制释放
		
		entity->Print();
	}
	

	{
		std::shared_ptr<Entity> e0;
		{
			std::shared_ptr<Entity> sharedEntity = std::make_shared<Entity>();
			std::weak_ptr<Entity> weakEntity = sharedEntity;
			//如果使用weak_ptr时不会增加 引用次数(软连接)
			
			//shared是可以复制的 会申请一部分内存叫做 控制内存
			e0 = sharedEntity;
		}
	}
	
	std:cin.get();
}
```

```cpp title:箭头操作符
#include <iostream>
#include <string>

class Entity
{
public:
	int x;
public;
	void Print() const { std::cout << "Hello!" << std::endl; }
};

class ScopedPtr
{
private:
	Entity* m_Obj;
public:
	ScopedPtr(Entity* entity)
		: m_Obj(entity)
	{
	}
	
	~ScopedPtr(Entity* ptr)
	{
		delete m_Obj;
	}
	
	//Entity* GetObject() { return m_Obj; }
	Entity* operator->()
	{
		return m_Obj;
	}
	
	const Entity* operator->() const //const版本
	{
		return m_Obj;
	}
};

struct Vector3
{
	float x, y, z;
	//float x, z, y; //虽然在结构体中是一样的 但对应的内存地址其实是不一样的
};

int main()
{
	ScopedPtr entity = new Entity;
	//entity.GetObject()->Print();
	
	entity->Print();
	
	int offset = (int)&((Vector3*)nullptr)->x;
	//Vec3类型空指针去 结构体中找x的内存地址 转为int赋值给offset
	std::cout << offset << std::endl;//此处输出为0
	
	int offset0 = (int)&((Vector3*)nullptr)->z;
	std::cout << offset0 << std::endl;//此时输出为8 就是z相对于x的偏移量是8
	
	std::cin.get();
}
```

```cpp title:函数指针
#include <iostream>

void HelloWorld(int a)
{
	std::cout << "Hello World! Value: " << a << std::endl;
}

int main()
{
	HelloWorld(1);
	
	//void(*function)();	//auto最终的样子
	auto function = HelloWorld; //相当于指针 获得函数的内存地址 &HelloWorld
	//函数只是cpu指令
	
	typedef void(*HelloWorldFunction)(int);
	HelloWorldFunction function = HelloWorld;	//也可以这么写 配合typedef
	
	function(8);
	
	std::cin.get();
}
```

```cpp title:函数指针
#include <iostream>
#include <vector>

void PrintValue(int value)
{
	std::cout << "Value: " << value << std::endl;
}	//这个函数做的操作不复杂 可以直接写在在ForEach中
	//可以用lambda函数代替

void ForEach(const std::vector<int> values, void(*func)(int))
{
	for (int value : values)
		func(value);
}

int main()
{
	std::vector<int> values = { 1, 5, 4, 2, 3 };
	ForEach(values, PrintValue); //函数的中嵌套调用
	
	ForEach(values, [](int value) { std::cout << "Value: " << std::endl; });
	
	std::cin.get();
}
```
