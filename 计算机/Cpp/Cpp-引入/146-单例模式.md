```ad-note
title: 单例模式

**一种组织一堆全局变量和静态函数的方法**（本质都是在单一的命名空间下）
不打算实例化一个特定类的多个实例
只想要他们的数据版本，还有这些操作方法也是对于这个类（结构体）的数据处理

例如：渲染器 是一个很全局的东西
```

```cpp title:单例模式
#include <iostream>

class Singleton //单例是不得有构造函数的
{
public:
	Singleton(const Singleton &) = delete; //保证构造函数是不允许存在的
	static Singleton& Get()
	{
		return s_Instance;
	}
	
	void  Function(){}
private:
	Singleton() {}
	static Singleton s_Instance; //定义一个指针
};

Singleton Singleton::s_Instance;

int main()
{
	Singleton& instance = Singleton::Get(); //这里就不能用简单的Singleton直接赋值
	//只能用引用的方式传地址 操作单例的类中Singleton数据
	
	Singleton::Get().Function(); //Get返回s_Instance 然后在单例上运行Function操作
	
	std::cin.get();
}
```

```cpp title:单例模式-案例
#include <iostream>

class Random //也可以用命名空间的 但是失去了类这种private public 还有引用的特性
{
public:
	Random(const Random &) = delete;

	{
		//static Random s_Instance; //可以这里构造一个static的Instance 后续都在内存中
		//只在第一次使用时创建 生命周期为整个程序
		return s_Instance;
	}
	
	float Float() { return m_RandomFenerator; }
	
	//static float Float() { return Get().IFloat(); } //这个是两步合一步 上面部分直接拉到private里面做调用
private:
	Random() {}
	float m_RandomFenerator = 0.5f;
	static Randomn s_Instance;
		//internal
	//float IFoat() { return m_RandomFenerator; } //拉进来
};

Random Random::s_Instance; //这里其实还要一个静态声明

int main()
{
	auto& random = Random::Get();
	float number = random.Float(); //这个是number的两部实现 下面是一步解决
	
	float number = Random::Get().Float(); //Random一般就是单例的
	
	std::cout << number << std::endl;
	
	std::cin.get();
}
```
