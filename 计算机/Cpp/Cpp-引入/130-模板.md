```cpp title:模板
#include <iostream>
#include <string>

/*
void Print(int value)
{
	std:cout << value << std:endl;
}

void Print(float value) //重载
{
	std:cout << value << std:endl;
}

void Print(std:string value) //重载
{
	std:cout << value << std:endl;
}
*/
template<typename T> //泛型 包含的类型即typename 改为class也行(避免混淆还是不要用了)
					//这个操作等价于上面仨等等等等类型
void Print(T value)
{
	std:cout << value << std:endl;
}//tips:如果main中没有调用Print 即使值为valu 编译照常通过(MSVC会通过 Clang会报错)
//使用template时 进行编译才会具体转化为代码


template<int N>

class Array
{
private:
	int m_Array[N];
public:
	int GetSize() const { return N; }
}

int main()
{
	Print<int>(5); //强制指定类型int
	
	Print("Hello");
	Print(5.5f);
	
	Array<5> array;
	//这个操作实际上就是
	/**/
	std:cout << array.GetSize() << std:endl;
	
}
```
