```cpp title:模板
#include <iostream>
#include <string>

/*
void Print(int value)
{
	std:cout << value << std:endl;
}

void Print(float value) //重载
{
	std:cout << value << std:endl;
}

void Print(std:string value) //重载
{
	std:cout << value << std:endl;
}
*/
template<typename T> //泛型 包含的类型即typename 改为class也行(避免混淆还是不要用了)

void Print(T value)
{
	std:cout << value << std:endl;
}

int main()
{
	Print<int>(5); //强制指定类型int
	
	Print("Hello");
	Print(5.5f);
}
```
