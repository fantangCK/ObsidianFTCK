```ad-note
title: 析构函数

波浪号~加上类的对象

用于清理手动分配的内存 超出作用域**自动调用析构函数**

```

```cpp title:析构函数

class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		X = 0.0f;
		Y = 0.0f;
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity() //析构函数
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}

void Function()
{ //栈上创建
	Entity e;
	e.Print();
	//e.~Entity(); //手动调用方法,不常用
} //作用域结束跳回Function()调用位置
//同时调用析构函数

int main()
{
	Function();
	std:cin.get();
}
```

```cpp title:作用域

class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		X = 0.0f;
		Y = 0.0f;
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity() //析构函数
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}

int* CreateArray()
{
	//int array[50]; //这样创建会因为作用域而被清理
	int array = new int[50]; //实际上在局部函数里面创建数组是很奇怪的
	return array;
}

void FillArray(int* array)
{
	//Do sth.
}

class ScopedPtr //作用域指针类
{
private:
	Entity* m_Ptr;
public:
	ScopedPtr(Entity* ptr)
		: m_Ptr(ptr)
	{
	}
	
	~ScopedPtr(Entity* ptr)
	{
		delete m_Ptr;
	}
}

}
int main()
{
	int* a = CreateArray();
	//一般不这么操作 实际操作如下
	
	int array[50];
	FillArray(array);
	
	{
	Entity e;
		//输出会有两个 "创建Entity" "销毁Entity"
	}
	
	{
		ScopedPtr e = new  Entity();
		Entity* e = new Entity(); //堆上创建的e不会因为作用域而消失
								//因而输出只有一个 "创建Entity"
	}

	std:cin.get();
}
```
