```ad-note
title: 左值与右值

左值有实际存储空间

注意
- ~~等号右边就是右值~~
- 右值不只是字面量 也可以是函数返回结果

```

```cpp title:左值与右值
#include <iostream>

int Getvalue()
{
	return 10;
}

int& Getvalue() //左值引用
{
	static int value = 10;
	return value;
}

void SetValue(const int& value) //同时支持了左值和右值
{
}

void PrintName(std::string& name) 
{ 
	std::cout << "[Lvalue]" << name << std::endl; //左值引用
}

void PrintName(std::string&& name) 
{
	std::cout << "[Rvalue]" << name << std::endl; //双引用 右值引用
}	//不用临时变量转换(有点偷资源的味道)
				
void PrintName(const std::string& name) 
{ 
	std::cout << name << std::endl; 
}	//这里const不仅仅向编译器保证函数不修改实参 还允许实参为右值(做了临时变量的转换)
	//最终还是左值引用

int main()
{
	int i = 10;
	//左值 右值
	int a = i;
	int i = GetValue();
	
	const int& a = 10; //特殊规则 等价于
	/*
	int temp = 10; //创建了一个临时的左值
	const int& a  = temp;
	*/
	GetValue() = 5;
	
	SetValue(i); //i是左值
	SetValue(10); //10是临时右值 传入的时候被作为左值
	
	std::string firstName = "Yan";
	std::string lastName = "Chernikov";
	//这个例子中 左边都是左值 右边都是右值
	//注意firstName + lastName作为临时变量 也为右值
	std::string fullName = firstName + lastName;
	PrintName(fullName);
	//PrintName(firstName + lastName); //这里为右值无法传参
	//所以很多C++写 常量引用
	std::cin.get();
}
```

```ad-note
title: 参数计算顺序

像是int value = 3 + 4 在编译时就完成计算得到值7
后自增运算++ 在编译过程中是被允许并行计算的 不必按照顺序对参数求值

注:
-  `c++17`之后要求后缀表达式必须在别的表达式之前被计算
- Clang Gcc都会对这种未定义行为报错 MSVC不会(就一完蛋玩意)
```

```cpp title:参数计算顺序
#include <iostream>

void PrintSum(int a ,int b)
{
	std::cout << a << "+" << b << "=" << (a + b) << std::endl;
}

int main()
{
	int value = 0;
	PrintSum(value++, value++); 
	/*	这里是未定义行为
	*	debug下 1 + 0 = 1
	*	release下 0 + 0 = 0
	*/
	PrintSum(++value, ++value); 
	/*	这里是未定义行为
	*	debug下 4 + 4 = 8
	*	release下 4 + 4 = 8
	*/
	std::cin.get();
}
```

```ad-note
title: 移动语义
简而言之 内存所有权的转移(移动对象 而不是复制)

注:
- 由于C++11引入的右值引用使得移动语义是可行的
- 参考[115-构造函数](115-构造函数.md) 中浅拷贝
```

```cpp title:移动语义
#include <iostream>

class String
{
public:
	String() = default; //默认构造函数
	String(const char* string)
	{
		printf("Created!\n");
		m_size = strlen(string);
		m_Data = new char[m_Size];
		memcpy(m_Data, string, m_Size);
	}
	
	String(const String& other)
	{
		printf("Copied!\n");
		m_size = other.m_Size;
		m_Data = new char[m_Size];
		memcpy(m_Data, other.m_Data, m_Size);
	}
	
	String(String&& other) noexcept //告诉编译器这里不需要抛异常
	{
		printf("Moved!\n");
		m_Size = other.m_Size;
		m_Data = other.m_Data; //浅拷贝
		
		other.m_Size = 0; //创建空对象 后面析构的时候就清理的是0 和nullptr
		other.m_Data = nullptr;
	}
	
	~String()
	{
		printf("Destroyed!\n");
		delete[] m_Data;
	}
	
	void Print()
	{
		for ( uint32_t i = 0; i < m_Size; i++ )
			printf("%c", m_Data[i])
		printf("\n");
	}
private:
	char* m_Data;
	uint32_t m_Size;
};

class Entity
{
public:
	Entity(const String& name) //这里是左右值都可以 右值会经过转换
		: m_Name(name) //这里多的一次拷贝
	{
	}
	
	Entity(String&& name)	//右值直接引用的构造函数(没有隐式转换)
		: m_Name(name) 
		//: m_Name(std::move(name)) //用move同理
	{
	}
	
	void PrintName()
	{
		m_Name.Print();
	}
private:
	String m_Name;
};

int main()
{
	Entity entity(String("Cherno")); //运行这里会创建 然后有一份Copy
	entity.PrintName();
	
	std::cin.get();
}
```

```cpp title:move
#include <iostream>

class String
{
public:
	String() = default;
	String(const char* string)
	{
		printf("Created!\n");
		m_size = strlen(string);
		m_Data = new char[m_Size];
		memcpy(m_Data, string, m_Size);
	}
	
	String(const String& other)
	{
		printf("Copied!\n");
		m_size = other.m_Size;
		m_Data = new char[m_Size];
		memcpy(m_Data, other.m_Data, m_Size);
	}
	
	String(String&& other) noexcept
	{		//偷资源
		printf("Moved!\n");
		m_Size = other.m_Size;
		m_Data = other.m_Data; //浅拷贝
		
		other.m_Size = 0;
		other.m_Data = nullptr;
	}
	//此处加上引用的构造
	String& operator=(String&& other) noexcept
	{	//这样就是没有创建新对象 把对象赋值给别名
		//防呆(防止把other是自身)写法
		if(this != &other)
		{
			printf("Moved!\n");
			//当前string对象为m_Data指向的 所以在此之前要清理指向 然后把m_Data指向改为dest
			delete[] m_Data;
			
			m_Size = other.m_Size;
			m_Data = other.m_Data; //确保不会误删已经被转移的资源
			
			other.m_Size = 0;
			other.m_Data = nullptr;
		}
		return *this;
	}
	~String()
	{
		printf("Destroyed!\n");
		delete[] m_Data;
	}
	
	void Print()
	{
		for ( uint32_t i = 0; i < m_Size; i++ )
			printf("%c", m_Data[i])
		printf("\n");
	}
private:
	char* m_Data;
	uint32_t m_Size;
};

class Entity
{
public:
	Entity(const String& name)
		: m_Name(name)
	{
	}
	
	Entity(String&& name)
		: m_Name(std::move(name))
	{
	}
	
	void PrintName()
	{
		m_Name.Print();
	}
private:
	String m_Name;
};

int main()
{
	String string = "Hello";
	//String dest = string; //这样就是简单的复制进去
	//String dest = (String&&)string;
	//强制转换为右值 string变为临时变量 被清理掉 将指向"Hello"的地址控制都给了dest
	//String dest((String&&)string); //和上面等价
	String dest = std::move(string); //和上面等价 但更优雅move自动找对应类型右值转换
	//这里其实还是创建了一份拷贝 涉及了一个 = 赋值运算符和assign等价
	//后续修增加了含=的移动赋值重载函数
	
	String apple = "Apple";
	String dest;
	
	std::cout << "Apple: ";
	apple.Print(); //此处输出为Apple
	std::cout << "Dest: ";
	dest.Print(); //此处没有移动赋值 输出为空
	
	dest = std::move(apple); //std::move(apple)转换为右值
	//由于 = 触发 移动赋值运算符 的重载函数
	std::cout << "Apple: ";
	apple.Print(); //移动语义后删除了apple
	std::cout << "Dest: ";
	dest.Print(); //此时Dest输出为Apple
	
	std::cin.get();
}
```
