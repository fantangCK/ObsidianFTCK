```ad-note
title: 多维数组

二维、三维、四维等等
实际上就是数组套娃
原理上就是二级指针 指向一级指针数组 也是套娃ಥ_ಥ
```

```cpp title:多维数组
#include <iostream>

int main()
{
	int* array = new int[50];	//这里分配200字节(32bit)
	
	int** a2d = new int*[50];	//32系统 4位 64位系统8位指针
	//a2d[0] = nullptr; //a2d才是int*指针
	
	
	//二维
	int** a2d = new int*[50];
	for (int i = 0; i < 500; i++)
	{
		a2d[i] = new int[50];	//原理
	}
	//注意由于内存碎片化 会导致cache miss 使得读写内存要比一维慢得多的多
	
	
	//三维
	int*** a3d = new int**[50];
	for (int i = 0; i < 500; i++)
	{
		a3d[i] = new int*[50];
		for (int j = 0; j < 50; j++)
		{
			a3d[i][j] = new int[50];
			//取[]其实就是两层解引用
			/*
			等价于
			int** ptr = a3d[i];
			ptr[j] = new int[50];
			*/
		}
	}
	//如果是堆分配 还需要遍历取delete内存
	delete[] array; //delete只能free一层
	
	for (int i = 0; i < 500; i++)
	{
		delete[] a2d[i];	
	}
	delete[] a2d;
	
	std::cin.get():
}
```
