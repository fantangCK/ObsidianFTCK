```cpp title:栈和堆内存区别

#include <iostream>
#include <string>
struct Vector3
{
	float x, y, z;
	
	Vector3()
		: x(10), y(11), z(12);
};

int main()
{
	int value = 5;
	int array[5];
	array[0] = 1;
	array[1] = 2;
	array[2] = 3;
	array[3] = 4;	//这几个数字在内存是反向存储的(像栈一样一个塞一个 然后指针往内存后移动)
	array[4] = 5;	//内存上这五个都是连续在一起的
	Vector3 vector; //vector上三个数也是在一起的
	//栈上
	
	int* hvalue new int;
	*hvalue = 5;
	harray[0] = 1;	//tips:批量加h shift+alt
	harray[1] = 2;
	harray[2] = 3;
	harray[3] = 4;
	harray[4] = 5;
	int* harray =  new int[5];
```
