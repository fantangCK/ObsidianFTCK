```cpp title:性能测试
#include <iostream>
#include <memory>

#include <chrono>

#include <array>

class Timer
{
public:
	Timer()
	{
		m_StartTimepoint = std::chrono::high_resoultion_clock::now();
	}
	
	Timer()
	{
		Stop();
	}
	
	void Stop()
	{
		auto endTimepoint = std::chrono::high_resoultion_clock::now();
		
		auto  start = std::chrono::time_point_cast<std::chrono::microseconds>(m_StartTimepoint).time_since_epoch().cout();
		//time_since_epoch 时间起始点到现在的时长
		auto  end = std::chrono::time_point_cast<std::chrono::microseconds>(endTimepoint).time_since_epoch().cout();
		//转换时间为 微秒
		auto duration = end - start;
		double ms = duration * 0.001
		
		std::cout << duration << "us (" << ms << "ms)\n";
	}
private:
	std::chrono::time_point::<std::chrono::high_resolution_clock> m_StartTimepoint;
};

int main()
{
	{
		Timer timer;
		//计时器超出作用范围自动析构后计算运行时间(不含编译时间)
		int value = 0;
		for (int i = 0; i < 1000000; i++)
			value +=2; //编译优化的时候是直接计算然后输出 (直接包含结果送给寄存器)
	}
	std::cout << value << std::endl;
	
	struct Vector2
	{
		float x, y;
	}
	
	std::cout << "Make Shared\n"
	{
		std::array<std::shared_ptr<Vector2>, 1000> sharedPtrs;
		Timer timer; //移动到这里 使其关注于性能部分代码
		for (int i = 0; i < sharedPtrs.size(); i++)
			sharedPtrs[i] = std::make_shared<Vector2>();
	}			//堆上直接分配,使用单一的内存块来存储指针本身和对象数据
	
	std::cout << "New Shared\n"
	{
		std::array<std::shared_ptr<Vector2>, 1000> sharedPtrs;
		Timer timer;
		for (int i = 0; i < sharedPtrs.size(); i++)
			sharedPtrs[i] = std::shared_ptr<Vector2>(new vector2());
	}			//堆上创建 自己管理内存释放
	
	std::cout << "Make Unique\n"
	{
		std::array<std::make_unique<Vector2>, 1000> sharedPtrs;
		Timer timer;
		for (int i = 0; i < sharedPtrs.size(); i++)
			sharedPtrs[i] = std::make_unique<Vector2>();
	}		//堆上直接分配,只有一个指针管理对象,没有引用次数开销
	

	__debugbreak();
}
```
