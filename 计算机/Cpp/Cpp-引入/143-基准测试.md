```cpp title:例子
#include <iostream>
#include <memory>

#include <chrono>

class Timer
{
public:
	Timer()
	{
		m_StartTimepoint = std::chrono::high_resoultion_clock::now();
	}
	
	Timer()
	{
		Stop();
	}
	
	void Stop()
	{
		auto endTimepoint = std::chrono::high_resoultion_clock::now();
		
		auto  start = std::chrono::time_point_cast<std::chrono::microseconds>(m_StartTimepoint).time_since_epoch().cout();
		//time_since_epoch 时间起始点到现在的时长
		auto  end = std::chrono::time_point_cast<std::chrono::microseconds>(endTimepoint).time_since_epoch().cout();
		//转换时间为 微秒
		auto duration = end - start;
		double ms = duration * 0.001
		
		std::cout << duration << "us (" << ms << "ms)\n";
	}
private:
	std::chrono::time_point::<std::chrono::high_resolution_clock> m_StartTimepoint;
};

int main()
{
	{
		Timer timer;
		//计时器超出作用范围自动析构后计算运行时间(不含编译时间)
		int value = 0;
		for (int i = 0; i < 1000000; i++)
			value +=2;
	}
	
	std::cout << value << std::endl;
	__debugbreak();
}
```
