```ad-note
title: 三目运算符

由冒号 `:` 和 `?` 构成
是if的一个语法糖

```

```cpp title:三目
#include <iostream>
#include <string>

static int s_Level = 1;
static int s_Speed = 2;

int main()
{
	if (s_Level > 5)
		s_Speed = 10;
	else
		s_Speed = 5;
	// 等价于
	s_Speed = s_Level > 5 ? 10 : 5;
	
	std::string otherRank; //中间变量
	if (s_Level > 10)
		otherRank = "Master";
	else
		otherRank = "Beginner";
	// 等价于
	std::string rank = s_Level > 10 ? "Master" : "Beginner";
	
	//嵌套
	s_Speed = s_Level > 5 ? 10 : 5;
	
	std::cin.get();
}
```
