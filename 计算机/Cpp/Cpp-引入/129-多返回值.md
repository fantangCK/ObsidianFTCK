```ad-note
title: 多返回值

C++默认是不能return两个值的

处理办法
- 传出一个结构体 数组 动态数组等等
- 传入用引用 然后直接对 传入值赋值(变向传出了多个值) 性能要比传出结构体要好

```

```cpp title:多返回值-结构体
#include <iostream>

// 定义结构体来存放多个值
struct Result {
    int value1;
    int value2; //这里是栈上建立
};

// 函数返回结构体
Result getValues() {
    Result res;
    res.value1 = 10;
    res.value2 = 20;
    return res;
}

int main() {
    Result result = getValues();
    std::cout << "Value1: " << result.value1 << ", Value2: " << result.value2 << std::endl;
}

```

```cpp title:多返回值-结构体-堆上

#include <iostream>
#include <memory>
using namespace std;

// 定义结构体
struct Data {
    int value1;
    double value2;
    string value3;
};

// 函数返回堆上分配的结构体的智能指针
unique_ptr<Data> getValues() {
    // 在堆上分配结构体
    unique_ptr<Data> data = make_unique<Data>();
    data->value1 = 10;
    data->value2 = 20.5;
    data->value3 = "Hello, Heap!";
    return data;
}

int main() {
    // 获取堆上分配的结构体智能指针
    unique_ptr<Data> result = getValues();
    
    // 使用返回的结构体
    cout << "Value1: " << result->value1 << ", Value2: " << result->value2 << ", Value3: " << result->value3 << endl;

    // 无需手动释放内存，unique_ptr 会自动管理
    return 0;
}


```

```cpp title:多返回值-引用
#include <iostream>
using namespace std;

// 函数通过引用参数传出多个值
void getValues(int &value1, int &value2) {
    value1 = 10;
    value2 = 20;
}

int main() {
    int val1, val2;
    getValues(val1, val2);
    std::cout << "Value1: " << val1 << ", Value2: " << val2 << std::endl;
    return 0;
}

```

```cpp title:多返回值-Vector

#include <iostream>
#include <vector>
using namespace std;

// 返回 vector
vector<int> getValues() {
    vector<int> values;
    values.push_back(10);
    values.push_back(20);
    return values;
}

int main() {
    vector<int> result = getValues(); //复制内存开销这里不考虑
    cout << "Values: ";
    
    for (int value : result) {
        cout << value << " ";
    }
    
    cout << endl;
    return 0;
}

```

```cpp title:多返回值-Tuple

#include <iostream>
#include <tuple>
using namespace std;

// 返回 tuple
tuple<int, double, string> getValues() {
    return make_tuple(10, 20.5, "Hello");
}

int main() {
    auto [val1, val2, val3] = getValues();  // 使用结构化绑定
    										//复制内存开销这里不考虑
    cout << "Value1: " << val1 << ", Value2: " << val2 << ", Value3: " << val3 << endl;
    return 0;
}


```
