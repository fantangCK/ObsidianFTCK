```cpp title:数组-栈上
#include <iostream>

int main()
{
	int example[5];
	int* ptr = example//数组其实就是一个指针
	
	example[0] = 2;
	example[1] = 6;
	//等价
	*(ptr + 1) = 6; //这里+1 是基于int 加四个字节
	*(int*)((char*)ptr + 4) = 6; //char类型指针+4字节再转换为int指针 令对应地址数为6
	
	example[-1] = 0;//debug模式会报错 但release不会 会导致内存访问超限
	
	std::cout << example[0] << std::endl;
	std::cout << example << std::endl; //打印整个数组
	
	for(int i = 0; i < 5;i++) //其实可以写作<=4 但考虑性能问题<5最佳
		example[i] = 2; //数组存储的数据在内存是连续的
		
	std::cin.get();	
}

```

```cpp title:数组-堆上
#include <iostream>
class Entity
{
	//内存间接寻址
public;
	static const int size = 5;
	int* example = new int[size];
	
	 Entity()
	 {
	 	for(int i = 0; i < 5 ; i++)
	 		example[i] = 2;
	 	int a[5];
	 	int count = sizeof(a) / sizeof(int); //5
	 	//sizeof只能告诉静态数组大小
	 } 
}


int main()
{
	int example[5]; //存在周期仅限于程序花括号内
	
	for (int i = 0; i < 5; i++)
		example[i] =2;
		
	int* another = new int[5]; //堆上申请内存
	for (int i = 0; i < 5; i++)
		another[i] =2;
		
	delete[] another;
	
	Entity e; //此时数组位置为一个指针 该指针指向一个内存地址 此地址才包含数组内的值
	//远端存储 影响性能
	
	std::cin.get();	
}

```
