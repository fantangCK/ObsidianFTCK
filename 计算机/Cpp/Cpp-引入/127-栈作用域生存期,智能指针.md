```cpp title:作用域

class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		X = 0.0f;
		Y = 0.0f;
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity() //析构函数
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}

int* CreateArray()
{
	//int array[50]; //这样创建会因为作用域而被清理
	int array = new int[50]; //实际上在局部函数里面创建数组是很奇怪的
	return array;
}

void FillArray(int* array)
{
	//Do sth.
}

class ScopedPtr //作用域指针类
{
private:
	Entity* m_Ptr;
public:
	ScopedPtr(Entity* ptr)
		: m_Ptr(ptr)
	{
	}
	
	~ScopedPtr(Entity* ptr)
	{
		delete m_Ptr;
	}
}

}
int main()
{
	int* a = CreateArray();
	//一般不这么操作 实际操作如下
	
	int array[50];
	FillArray(array);
	
	{
	Entity e;
		//输出会有两个 "创建Entity" "销毁Entity"
	}
	
	{
		Entity* e = new Entity(); //堆上创建的e不会因为作用域而消失
								//因而输出只有一个 "创建Entity"
	}
	
	{				//作用域指针类写法
		ScopedPtr e = new  Entity();
	}//超出作用范围自动析构 (并且是在堆上申请空间的)

	std:cin.get();
}
```
