```cpp title:作用域
#include <iostream>
#include <string>

class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		X = 0.0f;
		Y = 0.0f;
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity() //析构函数
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
};

int* CreateArray()
{
	//int array[50]; //这样创建会因为作用域而被清理
	int array = new int[50]; //实际上在局部函数里面创建数组是很奇怪的
	return array;
}

void FillArray(int* array)
{
	//Do sth.
}

class ScopedPtr //作用域指针类
{
private:
	Entity* m_Ptr;
public:
	ScopedPtr(Entity* ptr)
		: m_Ptr(ptr)
	{
	}
	
	~ScopedPtr(Entity* ptr)
	{
		delete m_Ptr;
	}
};

}
int main()
{
	int* a = CreateArray();
	//一般不这么操作 实际操作如下
	
	int array[50];
	FillArray(array);
	
	{
	Entity e;
		//输出会有两个 "创建Entity" "销毁Entity"
	}
	
	{
		Entity* e = new Entity(); //堆上创建的e不会因为作用域而消失
								//因而输出只有一个 "创建Entity"
	}
	
	{				//作用域指针类写法
		ScopedPtr e = new  Entity();
	}//超出作用范围自动析构 (并且是在堆上申请空间的)

	std:cin.get();
}

```

```ad-note
title: 智能指针

不用释放堆内存

智能指针有几个类型
- unique_ptr 指针无法复制和传递 
- shared_ptr 有一个引用计数 当引用计数为0时全部释放
- weak_ptr 被赋值等等操作 不会增加引用计数
```

```cpp title:智能指针
#include <iostream>
#include <string>

#include <memory>
class Entity
{
public:
	
	Entity()
	{
		std::cout << "Created Entity!" << std::endl;
	}
	
	~Entity()
	{
		std::cout << "Destroyed Entity!" << std::endl;
	}

};

int main()
{
	{
		//std::unique_ptr<Entity> entity(new Entity())
		//更好的做法
		std::unique_ptr<Entity> entity = std::make_unique<Entity>();
		
		//std::unique_ptr<Entity> e0 = entity; //不能这么做 因为一旦复制 等到作用域结束后 这个内存也会被强制释放
		
		entity->Print();
	}
	

	{
		std::shared_ptr<Entity> e0;
		{
			std::shared_ptr<Entity> sharedEntity = std::make_shared<Entity>();
			std::weak_ptr<Entity> weakEntity = sharedEntity;
			//如果使用weak_ptr时不会增加 引用次数(软连接)
			
			//shared是可以复制的 会申请一部分内存叫做 控制内存
			e0 = sharedEntity;
		}
	}
	
	std:cin.get();
}
```
