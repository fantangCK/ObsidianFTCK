```cpp title:构造函数-反例
class Entity
{
public:
	float X,Y;

	{
		X = 0.0f;

	{

{
	Entity e;
	e.Init();
	e.Print(); //每次运行都要初始化 很麻烦
	
	std:cin.get();
```

```cpp title:构造函数
class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		//X = 0.0f;
		//Y = 0.0f;
		//实际上这两句压根没用 只是当作初始化Init
	}
	
	Entity(float x , float y)//同名函数-函数重载
	{
		X = x;
		Y = y;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}

class Log
{
private:
	Log(){}; //隐藏构造函数
public:
	static void Write(){}
}

int main()
{
	Entity e(10.0f,5.0f);
	e.Print();
	
	Log::Write(); //不创建实例方法
	Log l; //默认构造函数
	//如果Log是private就无法访问
	
	std:cin.get();
}
```

```cpp title:堆与栈的初始化
#include <iostream>
#include <string>
using String = std::string;
 
class Entity
{
private;
	String m_Name;
public:
	Entity() : m_Name("Unknown") {};
	Entity(const String& name) : m_Name(name) {};

	const String& GetName() const { return m_Name;}
};

void Function()
{
	Entity entity = Entity("Cherno");
	//当调用此函数时 会新建一个栈上Entity 达到函数生命周期后(花括号位置)就会被回收
}

int main()
{
	Function();
	
	Entity entity;	//之所以没有参数不会导致内存溢出 是因为有默认的方法Entity()
	std::cout << entity.GetName() << std::endl;
	
	//为了扩展更多entity 在堆上声明方法如下(栈上一般就1-2M空间)
	Entity* entity = new Entity("Cherno");
	//new了就要清理
	std::cout << entity -> GetName() << std::endl;
	delete entity;
	std::cin.get();
}

```

```ad-note
title: 复制和拷贝构造函数

主要是解决,在堆上对象的拷贝引起的 `内存申请` 和 `指针指向`问题

有默认的拷贝构造函数 用于修改 具体参考下面的代码

```

```cpp title:复制
#include <iostream>
#include <string>

struct Vector2
{
	float x, y;
}

int main()
{
	Vector2* a = new Vector2();
	Vector2* b = a; //复制是指针 而不是值
	b++; //这个操作不会改变a 即改变的是指针本身
	
	b->x = 2;//但是 这个操作会同时改变a b 即改变的是指针对应的元素
	
	std::cin.get();
}
```

```cpp title:拷贝构造函数
#include <iostream>
#include <string>

class String
{
private:
	char* m_Buffer;
	unsigned int m_Size;
public:
	String(const char* string)
	{
		m_Size = strlen(string);
		m_Buffer = new char[m_Size + 1];
			//此处加一是为了包含空字符的截至符
		/*
		for (int i = 0; i < m_Size; i++)
			m_Buffer[i] = string[i];
		*/ //以下是简洁写法
		memcpy(m_Buffer, string, m_Size + 1);
		//m_Buffer[m_Size] = 0; //也可以自己在最后加0
	}
	
	String(const String& other); //默认的拷贝构造函数
		: m_Size(other.m_Size) //浅拷贝
	{
		//std::cout << "Copied String!" << std::endl;
		//检查拷贝构造函数
		m_Buffer = new char[m_Size + 1];
		memcpy(m_Buffer, other.m_Buffer, m_Size + 1);
		//这个才是深拷贝 分配新的内存空间
	}
	
	/*
		: m_Buffer(other.m_Buffer), m_Size(other.m_Size) {}; //或者这种
	*/
	
	/*
	{
		memcpy(this, &other, sizeof(String));
	} //这个也是一个内在运行逻辑
	*/
	//将other的指针指向的内存复制
	
	//String(const String& other)= delete;
	//如果强制不得使用复制构造函数这样写
	
	~String() //使用了new就要考虑释放 这里写了一个析构函数
	{
		delete[] m_Buffer;
	}
	
	char& operator[](unsigned int index)
	{
		return m_Buffer[index];
	}
	
	friend std::ostream& operator<<(std::ostream& stream, const String& string) //允许此操作访问private 友元friend操作
};

std::ostream& operator<<(std::ostream& stream, const String& string)
{
	//stream << string.GetBuffer(); //可以写个函数访问
	//直接访问private写法如下
	stream << string.m_Buffer;
	
	return stream;
}

void PrintString(const String& string)	
//如果直接String string 会有三次拷贝构造函数
{
	//String copy = string; //如果还是要复制值而不是引用
	std::cout << string << std::endl;
}

int main()
{
	String string = "Cherno";
	
	//String second = string;
	//此处需要再次分配 m_Buffer 即"深拷贝" 拷贝构造函数
	
	
	second[2] = 'a'; //此操作符要有一个重载函数
	//修正前:
	//包含了"浅拷贝" 因为这个拷贝操作 两个指针的地址一致了
	//此时会导致两次调用析构函数 但是析构函数在第一次就已经释放了这个内存(两个指针都无效了) 再次释放会报错
	
	PrintString(string);
	PrintString(second);
	
	std::cin.get();
}
```
