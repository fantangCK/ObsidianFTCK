```cpp title:构造函数-反例
class Entity
{
public:
	float X,Y;
	
	void Init()
	{
		X = 0.0f;
		Y = 0.0f;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}
int main()
{
	Entity e;
	e.Init();
	e.Print(); //每次运行都要初始化 很麻烦
	
	std:cin.get();
}
```

```cpp title:构造函数
class Entity
{
public:
	float X,Y;
	
	Entity()
	{
		//X = 0.0f;
		//Y = 0.0f;
		//实际上这两句压根没用 只是当作初始化Init
	}
	
	Entity(float x , float y)//同名函数-函数重载
	{
		X = x;
		Y = y;
	}
	
	void Print()
	{
		std:cout << X << "," << Y <<std::endl;
	}
}

class Log
{
private:
	Log(){}; //隐藏构造函数
public:
	static void Write(){}
}

int main()
{
	Entity e(10.0f,5.0f);
	e.Print();
	
	Log::Write(); //不创建实例方法
	Log l; //默认构造函数
	//如果Log是private就无法访问
	
	std:cin.get();
}
```

```cpp title:堆与栈的初始化
#include <iostream>
#include <string>
using String = std::string;
 
class Entity
{
private;
	String m_Name;
public:
	Entity() : m_Name("Unknown") {};
	Entity(const String& name) : m_Name(name) {};

	const String& GetName() const { return m_Name;}
};

void Function()
{
	Entity entity = Entity("Cherno");
	//当调用此函数时 会新建一个栈上Entity 达到函数生命周期后(花括号位置)就会被回收
}

int main()
{
	Function();
	
	Entity entity;	//之所以没有参数不会导致内存溢出 是因为有默认的方法Entity()
	std::cout << entity.GetName() << std::endl;
	
	//为了扩展更多entity 在堆上声明方法如下(栈上一般就1-2M空间)
	Entity* entity = new Entity("Cherno");
	
	std::cin.get();
}
```
