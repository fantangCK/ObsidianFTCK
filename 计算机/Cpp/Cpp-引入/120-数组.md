```cpp title:数组
#include <iostream>


int main()
{
	int example[5];
	
	example[0] = 2;
	example[1] = 6;
	
	example[-1] = 0;//debug模式会报错 但release不会 会导致内存访问超限
	
	std::cout << example[0] << std::endl;
	std::cout << example << std::endl; //打印整个数组
	
	for(int i = 0; i < 5;i++) //其实可以写作<=4
	
	std::cin.get();	
}
```
