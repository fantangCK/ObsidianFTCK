```ad-note
title: static 静态

类比class(类)中的private
主要体现在Linking阶段

```

```cpp title:"static例子-Static.cpp"

int s_Variable1 = 5;
static int s_Variable = 10; //这个变量就只存在此文件中,在Linking阶段也找不到

static void Function()

```

```cpp title:"static例子-Main.cpp"
#include <iostream>

int s_Variable1 = 10;
//会报错 重复定义了

extern int s_Variable; 
//会报错因为是s_Variable是static

void Function //函数同理

{
	std::cout << s_variable << std::endl;
	std::cin.get();
}
```

```ad-note
title: 静态对象在结构体的作用

- 每个声明的结构体实例 在使用static静态方法的时候**都要导入对应的实例** 否则找不到对应
- 静态变量同理,即全局性改变


```

```cpp title:"结构体中的static-1"
struct Entity
{
	static int x, y;
	
	void Print()
	{
		std::cout << x << "," << y << std::endl;
	}
}

int Entity::x;
int Entity::y;

int main()
{
	Entity e;
	e.x = 2;
	e.y = 3;

	Entity e1;
	e1.x = 5;
	e1.y = 8; 	
	//由于是static e和e1的x,y指向地址一致

	//上面四个赋值都等价于以下两个赋值
	Entity::x =5;
	Entity::y =8;
	
	e.Print();
	e1.Print();
	
	std::cin.get();
}

```

```cpp title:"结构体中的static-2"
struct Entity
{
	int x, y;
	
	static void Print()
	{
		std::cout << x << "," << y << std::endl;
	}
	//静态对象不能访问非静态变量 每个实例(instance)的变量都不一样
	//会报错
}

static void Print(Entity e)
{ //这种处理方法没问题,因为传入了实例e
	//非静态方法在编译的时候的样子
	std::cout << e.x << "," << e.y << std::endl;
}

int main()
{
	Entity e;
	e.x = 2;
	e.y = 3;

	Entity e1;
	e1.x = 5;
	e1.y = 8; 	
	
	Entity::Print();
	
	std::cin.get();
}

```

```ad-note
title: local static 局部静态

局部静态 相比于 静态只能在函数内部方位

局部静态的存在周期相当于程序的运行周期 可以直接的去访问

```

```cpp title:局部静态-函数用法
#include <iostream>

void Function()
{
	int i = 0; //用这个声明 运行时每次都是1 每次创建的函数都需要重新初始化一个i
	static int i = 0;
	i++;
	std::cout << i << std::endl;
}

int main()
{
	Function();
	Function();
	Function(); 
	std::cin.get();
}

```


```cpp title:指针变量引用的方法
#include <iostream>

class Singleton
{
private:
	static Singleton* s_Instance; //定义一个指针
public:
	static Singleton& Get(){ return *s_Instance; } //返回一个指针地址
	
	void Hello(){} //成员函数
};

Singleton* Singleton::s_Instance = nullptr; //实例的外部定义 初始化

int main()
{
	Singleton::Get().Hello(); //得到Sigleton的地址再执行的Hello
	//Hello不能脱离Get单独运行
	std::cin.get();
}

```

```cpp title:简洁写法
#include <iostream>

class Singleton
{
public:
	static Singleton& Get()
	{
		static Singleton instance; //static是第一次创建后就保持在内存
		return instance;
	} //返回一个指针
	
	void Hello(){} //成员函数
};

int main()
{
	Singleton::Get().Hello(); //得到Sigleton的地址再执行的Hello
	//Hello不能脱离Get单独运行
	std::cin.get();
}
```
