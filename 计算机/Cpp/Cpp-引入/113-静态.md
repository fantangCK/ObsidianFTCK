
```cpp title:"static例子-Static.cpp"

int s_Variable1 = 5;

```

```cpp title:"static例子-Main.cpp"
#include <iostream>

int s_Variable1 = 10;
//会报错 重复定义了

void Function //函数同理
```

```cpp title:"结构体中的static-1"
struct Entity
{
	static int x, y;
	
	void Print()
	{
		std::cout << x << "," << y << std::endl;
	}
}

int Entity::x;
int Entity::y;

int main()
{
	Entity e;
	e.x = 2;
	e.y = 3;

	Entity e1;
	e1.x = 5;
	e1.y = 8; 	
	//由于是static e和e1的x,y指向地址一致

	//上面四个赋值都等价于以下两个赋值
	Entity::x =5;
	Entity::y =8;
	
	e.Print();
	e1.Print();
	
	std::cin.get();
}

```

```cpp title:"结构体中的static-2"
struct Entity
{
	int x, y;
	
	static void Print()
	{
		std::cout << x << "," << y << std::endl;
	}
	//静态对象不能访问非静态变量 每个实例(instance)的变量都不一样
}

int main()
{
	Entity e;
	e.x = 2;
	e.y = 3;

	Entity e1;
	e1.x = 5;
	e1.y = 8; 	
	
	Entity::Print();
	
	std::cin.get();
}
```
