```cpp title:变量
#include <iostream>

int main(){
	int var = 8 
	char,short,long,long long //等等类型
	unsigned int //去除符号位 可以表示更大的数
	char a = 'A'; //ASCII表示的65(十六进制位41H)
	//char 类比character 字符
	std::cout << a << std:endl;
	
	float b = 5.5; //默认为double类型
	float b = 5.5f;//使用f或者F才表示为float类型 4字节类型
	
	bool variable = false;
	//占用一个字节 虽然可以用一个位来表示 但寻址用一个字节更方便
	//bool类型 除了0之外 都为ture
	
	std::cout << sizeof(double) << std:endl;
	//sizeof 输出占用给的 字节多少
	
}
```
