```cpp title:线程
#include <iostream>
#include <thread>

static bool s_Finished = false;

void DoWork()
{
	using namespace std::literals::chrono_literals; //chrono库提供了时间的表示
	
	std::cout << "Started thread id=" << std::this_thread::get_id() << std::endl;
	
	while (!s_Finished)
	{
		std::cout << "Working..." << std::endl;
		std::this_thread::sleep_for(1s);	//每秒运行一次 (这里使用1s即chrono库)
	}
}

int main()
{
	std::thread worker(DoWork);	//一直都在DoWork
	
	std::cin.get(); //待输入后 将s_Finished赋值为1 停掉进程
	s_Finished = true;
	
	worker.join(); //阻塞进程 等待并发的进程全部结束后 再继续主进程
	td::cout << "Finished." << std::endl;
	
	std::cin.get();
}
```
