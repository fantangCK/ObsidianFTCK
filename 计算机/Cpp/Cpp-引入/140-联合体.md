```ad-note
title: 联合体

联合体只能有一个成员变量 适合处理名称一样

float a, b, c, d其对应的内存地址都是一样的

```

```cpp title:例子
#include <iostream>

struct Vector2
{
	float x ,y;
};

struct Vector4
{
	union //这个是匿名函数
	{
		struct
		{
			float x, y, z, w;
		};
		struct
		{
			Vector2 a, b; //a和x y内存空间一样;b和z w内存空间一样
		};
	};
};

void PrintVector2(const Vector2& vector)
{
	std::cout << vector.x << ", " << vector.y << std::endl;
}

int main()
{
	struct Union{
		union
		{
			float a;
			int b;
		};
	};
	
	Union u;
	u.a = 2.0f;
	
	std::cout << u.a << ", " << u.b << std::endl; //u.b会用浮点数输出表示
	
	
	Vector4 vector = { 1.0f, 2.0f, 3.0f, 4.0f };
	PrintVector2(vector.a);
	PrintVector2(vector.b);
	vector.z = 500.0f;
	std::cout << "-----------------------------" << std::endl;
	PrintVector2(vector.a);
	PrintVector2(vector.b);
	
	std::cin.get();
}
```
