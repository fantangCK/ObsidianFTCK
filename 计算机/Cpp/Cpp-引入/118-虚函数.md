#include <iostream>
#include <string>

class Entity
{
public:
	std::string getName(){ return "Entity"; }
}

class Player : public Entity
{
private:
	std::string m_Name;
public:
	Player(const std::string& name)
		: m_Name(name) {} //初始化

	std::string getName() { return m_Name; }
}

void PrintName(Entit* entity)
{
	std::cout << entity->getNmae() << std::endl;
}

int main()
{
	Entity* e = new Entity(); //堆上声明的Entity
	PrintName(e);
	
	Player*p = new Player("Cherno")
	PrintName(p);
	
	std::cin.get();
}
