```cpp title:虚函数
#include <iostream>
#include <string>

class Entity
{
public://虚函数 virtual表(故有一定内存开销)
	virtual std::string getName(){ return "Entity"; }
};

class Player : public Entity
{
private:
	std::string m_Name;
public:
	Player(const std::string& name) //构造函数
		: m_Name(name) {} //成员变量初始化 相比于声明再赋值更高效

	std::string getName() override { return m_Name; } //c++11引入 增加可读性
						//复写的虚函数
	//std::string getName() { return m_Name; }
}

void PrintName(Entity* entity) 
{
	std::cout << entity->getNmae() << std::endl;
}

int main()
{
	Entity* e = new Entity(); //堆上声明的Entity
	PrintName(e);
	
	Player*p = new Player("Cherno")
	PrintName(p);//由于此处 函数的接口参数为父级Entity方法 输出的是Entity
	//使用虚函数后 Entity* entity 自动变为Player* entity
	//调用的也就是Player类中的函数
	
	std::cin.get();
}
```

```ad-note
title: 纯虚函数-接口

## 1. 什么是纯虚函数？

纯虚函数是一种在基类中声明但不提供实现的函数，用于定义接口规范。它要求所有派生类必须实现该函数，否则派生类也将成为抽象类，无法实例化。

---

## 2. 抽象类

包含至少一个纯虚函数的类被称为抽象类。抽象类不能被直接实例化，只能作为基类供其他类继承使用，确保派生类实现特定接口。

---

## 3. 纯虚函数的作用

1. **定义接口：** 提供统一的接口规范，强制派生类实现特定功能。
2. **支持多态：** 允许通过基类指针或引用调用派生类的实现，支持运行时多态。
3. **代码复用：** 通过接口和抽象类实现代码解耦，提升系统的可扩展性和可维护性。

---

## 4. 接口类的特点

接口类通常只包含纯虚函数和（可选的）虚析构函数，具有以下特点：
- **不能实例化：** 只能作为接口被继承使用。
- **强制实现：** 派生类必须实现接口中所有的纯虚函数。
- **多态支持：** 允许不同派生类通过相同接口进行调用，实现多态行为。

接口类广泛应用于设计模式和面向对象编程中，用于定义系统中的标准操作接口。

```

```cpp title:纯虚函数
#include <iostream>
#include <string>


class Printable
{
public:
	virtual std::string getClassName() = 0; //有一个纯虚函数
}

class Entity : public Printable
{
public:
	virtual std::string getName() { return "Entity"; }
	std::string getClassName() override { return "Entity";}
};

class Player : public Entity
{
private:
	std::string m_Name;
public:
	Player(const std::string& name)
		: m_Name(name) {}

	std::string getName() override { return m_Name; }
	//std::string getClassName() override { return "Player";}
	//如果Player没有实现getClassName，则会调用父类Entity的getClassName方法
}

class A : public Printable
{
public:
	std::string getClassName() override { return "A";}
};

void Print(Printable* obj) //不在乎具体什么类
{
	std::cout << obj->getClassName() << std::endl;
}

int main()
{
	//Entity* e = new Printable(); //不成立 需要一个子类才能生成
	Entity* e = new Entity();
	
	Player*p = new Player("Cherno");
	
	Print(e);
	Print(p);
	Print(new A()); //实例,不要这么写,会有内存溢出
	//如果类中有实现Print的方法 即类中有getClassName()才能正常返回
	
	std::cin.get();
}
```
