```cpp title:类

class Player{
{
public:
	int x, y;
	int speed;
	
	void Move1(int xa, int ya) //在类里面就不必再调用Player
	{
	player.x += xa * player.speed;
	player.y += ya * player.speed;
	}
};

void Move(Player& player, int xa, int ya)
{
	player.x += xa * player.speed;
	player.y += ya * player.speed;
}

int main(){
	Player player; //声明了一个对象
	player.x = 5;
	Move(player, 1, -1);
	
	player.Move(1, -1);
}
```

```cpp title:类常见写法
class Log
{
public: //变量函数都可以分开写
	const int LogLevelError = 0;
	const int LogLevelWarning = 1;
	const int LogLevelInfo = 2;
private:
	int m_LogLevel = LogLevelInfo;
	//m 指member 成员变量 -写法约定
public:
	void SetLevel(int level)
	 {
	 	m_LogLevel = level;
	 }
	 
	 void Warn(const char* message)
	 {
	 	std::cout << message << std::endl;
	 }
};

int  main()
{
	Log log; //初始化
	//先明确需求再去补充clas内容
	log.SetLevel(LogLevelWarning); 
	log.warn("hello!");
	std::cin.get();
}
```
