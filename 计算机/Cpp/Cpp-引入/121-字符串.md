```cpp title:基础char字符串
#include <iostream>

int main()
{
	const char* name = "Cherno"; //const目的是固定
	//虽然有*但不是在堆上申请的内存 只有new是(用到new就要用到delete) 
	char name2[7] ={ 'C', 'h', 'e', 'r', 'n', 'o', '\0' }
	//使用'\0' 或者 0 都可以
	//此处不加\0的话就cout就不知道在何处停止输出 而在debug模式下 在数组内存后面会跟着 数组"守卫" 用于提示访问到溢出区域
	std::cout << name2 << std::endl;
	
	//内存中Cherno内存后面跟着有一个空终止符
	name[2] = 'a';//const下不能修改 删去就行
}
```

```cpp title:基础string字符串
#include <iostream>

	std::cout << string << std::endl;

{
	std::string name = "Cherno";//用"" 直接赋值默认是给与const *隐式转换*
	
	std::string name = std::string("Cherno") + " hello!"; //这样操作是可以的 不再是指针相加
	//strlen strcpy等等功能在string中都有
	name.size(); //获取name长度
	name += " hello!"; //std::string name = "Cherno" + "hello!"是不可以的
	//+=操作在string中被重载
	
	bool contains = name.find("no") != std::string::nps;
									//std::string::nps变量:不存在的地方
					//返回no所在首位置
	std::cout << name << std::endl;
	
}
```

```ad-note
title: 字面量

字面量的内存永远在内存的只读区域

<sub>**例:**在编译阶段像是“test”赋值给name数组之后 对name[1]操作 test本身常量没有变化 而是name的变量发生变换</sub>
```

```cpp title:字面量
#include <iostream>
#include <string>

int main()
{
	"Cherno"; //默认是const char [7] *为什么7不再做解释*
```
