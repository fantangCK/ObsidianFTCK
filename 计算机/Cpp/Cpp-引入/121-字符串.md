```cpp title:基础char字符串
#include <iostream>

int main()
{
	const char* name = "Cherno"; //const目的是固定
	//虽然有*但不是在堆上申请的内存 只有new是(用到new就要用到delete) 
	char name2[7] ={ 'C', 'h', 'e', 'r', 'n', 'o', '\0' }
	//使用'\0' 或者 0 都可以
	//此处不加\0的话就cout就不知道在何处停止输出 而在debug模式下 在数组内存后面会跟着 数组"守卫" 用于提示访问到溢出区域
	std::cout << name2 << std::endl;
	
	//内存中Cherno内存后面跟着有一个空终止符
	name[2] = 'a';//const下不能修改 删去就行
}

```

```cpp title:基础string字符串
#include <iostream>
#inlcude <string>

int main()
{
	std::string name = "Cherno";//用"" 直接赋值默认是给与const *隐式转换*
	//strlen strcpy等等功能在string中都有
	name.size(); //获取name长度
	name += "hello!"; //std::string name = "Cherno" + "hello!"是不可以的
	std::cout << name << std::endl;
	
}
```
