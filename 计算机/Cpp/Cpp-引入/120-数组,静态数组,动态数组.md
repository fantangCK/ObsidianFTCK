```cpp title:数组-栈上
#include <iostream>

int main()
{
	int example[5];
	int* ptr = example//数组其实就是一个指针
	
	example[0] = 2;
	example[1] = 6;
	//等价
	*(ptr + 1) = 6; //这里+1 是基于int 加四个字节
	*(int*)((char*)ptr + 4) = 6; //char类型指针+4字节再转换为int指针 令对应地址数为6
	
	example[-1] = 0;//debug模式会报错 但release不会 会导致内存访问超限
	
	std::cout << example[0] << std::endl;
	std::cout << example << std::endl; //打印整个数组
	
	for(int i = 0; i < 5;i++) //其实可以写作<=4 但考虑性能问题<5最佳
		example[i] = 2; //数组存储的数据在内存是连续的
		
	std::cin.get();	
}

```

```cpp title:动态数组
#include <iostream>
#include <string>
#include <vector>

struct Vertex
{
	float x, y, z;
};

std:ostream&operator<<(std:ostream&stream,const Vertex&vertex)
{
	stream << vertex.x << "," << vertex.y << "," <<vertex.Z;
	return stream;
}

void Function(const std::vector<Vertex>& vertices) {}
										//尽量使用引用&
										
int main()
{
	//Vertex* vertices = new Vertex[50000000];  
	//为了每次都增加顶点 需要在使用前直接申请一大块区域 但是这也造成大量内存浪费
	
	std::vector<Vertex> vertices;
	//如果这样设计 内存是在一条线上分布(好处:读写很快 坏处:扩展空间的时候 需要反复拷贝,移动内存)
	
	//std::vector<Vertex*> vertices;
	//在存储字符型的话采用指针较优 如果涉及扩展 即使内存是碎片的 也比整段拷贝移动要好
	
	//总之 如何设计看需求(选用栈存储 / 指针堆存储)
	
	vertices.push_back({ 1, 2, 3 }); //别的语言可能叫add等等 加入
	vertices.push_back({ 4, 5, 6 });
	
	
	for (int i = 0; i < vertices.size(); i++)
		std::cout << vertices[i] <<std::endl;
	 //两种写法 更推荐下面的
	 
	 vertices.erase(vertices.begin() + 1);
	 
	for (const Vertex& v: vertices)
		std::cout << vertices[i] <<std::endl;
	//加const没啥必要 但是这里的引用& 避免了复制占用内存
	std::cout << v <<std::endl;
	
	vertices.clear(); //清除
	
	std:cin.get();
}
```

```ad-note
title: Vector使用优化

push_back如果用完了这段内存空间 会重新调整大小 重新分配
即 避免复制对象

```

```cpp title:vector优化
#include <iostream>
#include <string>
#include <vector>

struct Vertex
{
	float x, y, z;
	
	Vertex(float x, float y, float z)
		: x(x), y(y), z(z)
	{
	}
	
	Vertex(const Vertex& vertex)
		: x(vertex.x), y(vertex.y), z(vertex.z)
	{
		std::cout << "Copied!" <<std::endl; //debug used
	}
	
};

int main()
{
	std::vector<Vertex> vertices;
	
	vertices.reserve(3): //调整(预留)空间为3 这里和构造函数中直接传入3不一样
	
	vertices.push_back(Vertex( 1, 2, 3 )); //main函数中调用了Vertex创建了新的vector
	vertices.push_back(Vertex( 4, 5, 6 )); //由于调整Vector大小也会多一份vector
	vertices.push_back(Vertex( 7, 8, 9 ));
	//不优化总计1 + (1 + 1) + (1 + 2) =6 次拷贝构造函数
	
	vertices.emplace_back(1, 2, 3); //这样是让vertices使用1,2,3参数直接赋值给vector
	vertices.emplace_back(4, 5, 6);
	vertices.emplace_back(7, 8, 9);
	//最后 一份拷贝函数都没有了
	
	std:cin.get();
}
```
